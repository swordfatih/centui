if(NOT CENTYPO_SYSTEM_FONTS)
    target_sources(centypo PRIVATE fonts.cpp)
elseif(WIN32)
    target_sources(centypo PRIVATE windows.cpp)
    set_source_files_properties("${CMAKE_CURRENT_SOURCE_DIR}/windows.cpp"
        TARGET_DIRECTORY centypo PROPERTIES COMPILE_DEFINITIONS NOMINMAX)
    target_link_libraries(centypo PRIVATE dwrite)
elseif(APPLE)
    target_sources(centypo PRIVATE apple.cpp)
    target_link_libraries(centypo PRIVATE "-framework CoreText" "-framework CoreFoundation")
elseif(ANDROID)
    target_sources(centypo PRIVATE android.cpp)
else()
    find_package(Fontconfig 2.12 REQUIRED)
    target_sources(centypo PRIVATE fontconfig.cpp)
    target_link_libraries(centypo PRIVATE Fontconfig::Fontconfig)
endif()
