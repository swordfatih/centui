function(centui_smoke name platform renderer)
    add_executable(centui_smoke_${name} smoke.cpp)
    target_compile_definitions(centui_smoke_${name} PRIVATE CENTUI_SMOKE_${name})
    target_link_libraries(centui_smoke_${name} PRIVATE ${platform} ${renderer})
    centui_font_asset(centui_smoke_${name})
    add_test(NAME smoke_${name} COMMAND centui_smoke_${name})
    set_tests_properties(smoke_${name} PROPERTIES TIMEOUT 30 LABELS integration)
endfunction()
if(CENTUI_BUILD_SDL)
    centui_smoke(SDL centui::sdl centui::sdl_renderer)
    if(CENTUI_BUILD_BGFX)
        centui_smoke(SDL_BGFX centui::sdl centui::bgfx)
    endif()
endif()
if(CENTUI_BUILD_SFML)
    centui_smoke(SFML centui::sfml_platform centui::sfml)
endif()
if(CENTUI_BUILD_GLFW AND CENTUI_BUILD_BGFX)
    centui_smoke(GLFW_BGFX centui::glfw centui::bgfx)
endif()
